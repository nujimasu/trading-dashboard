================================================================
 TradingView Pine Script v6 — RR1.5〜2.0 押し目反転
================================================================

使い方:
  1. TradingViewのPineエディタへ、下のコード部分を貼り付ける。
  2. 日足チャートで使用する。
  3. SPY/QQQ地合い、個別トレンド、押し、反転条件を満たすとシグナル表示。
  4. 本番運用では翌日寄り付き直後に表示されたOpen RRを確認し、
     1.5〜2.0の範囲だけを成行エントリーする。

重要な差異:
  ・TradingViewの通常の履歴バックテストでは「翌日始値を確認して、その同じ始値で
    新規注文を約定」する処理を完全には再現できない。
  ・下記strategyは前日終値基準RRが1.5〜2.0の時に注文し、次の始値で約定する近似版。
  ・正確な本番条件は、翌日始値で openRR を再計算して範囲外なら見送ること。
  ・ダッシュボード側の全米株横断ランキング、最大30銘柄、既存open件数管理は
    TradingView単一チャートでは再現しない。
  ・決算日除外は含まない。

----------------------------------------------------------------
 Pine Scriptコード
----------------------------------------------------------------

//@version=6
strategy("RR1.5-2.0 Pullback Reversal", overlay=true, pyramiding=0,
     initial_capital=100000, commission_type=strategy.commission.percent,
     commission_value=0.10, process_orders_on_close=false,
     calc_on_order_fills=true)

// ----- Inputs -----
minRR          = input.float(1.5, "Minimum RR", step=0.1)
maxRR          = input.float(2.0, "Maximum RR", step=0.1)
stopLookback   = input.int(20, "Stop lookback", minval=5)
targetLookback = input.int(60, "Resistance lookback", minval=20)
atrBuffer      = input.float(0.1, "ATR buffer", step=0.05)
maxHolding     = input.int(30, "Maximum holding bars", minval=1)
minVolumeRatio = input.float(0.8, "Minimum volume ratio", step=0.05)
minPaCount     = input.int(4, "Minimum PA count", minval=1, maxval=7)
minOscCount    = input.int(1, "Minimum oscillator count", minval=1, maxval=3)

// ----- Indicators -----
ema20  = ta.ema(close, 20)
ema50  = ta.ema(close, 50)
ema200 = ta.ema(close, 200)
atr14  = ta.atr(14)
rsi14  = ta.rsi(close, 14)
vol20  = ta.sma(volume, 20)
volRatio = volume / vol20

[macdLine, macdSignal, macdHist] = ta.macd(close, 12, 26, 9)
stochK = ta.stoch(close, high, low, 14)
stochD = ta.sma(stochK, 3)

// ----- Strict SPY / QQQ regime -----
[spyClose, spyEma50, spyEma200] = request.security("AMEX:SPY", "D",
     [close, ta.ema(close, 50), ta.ema(close, 200)],
     barmerge.gaps_off, barmerge.lookahead_off)
[qqqClose, qqqEma50, qqqEma200] = request.security("NASDAQ:QQQ", "D",
     [close, ta.ema(close, 50), ta.ema(close, 200)],
     barmerge.gaps_off, barmerge.lookahead_off)

marketOk = spyClose > spyEma200 and spyEma50 > spyEma200 and
     qqqClose > qqqEma200 and qqqEma50 > qqqEma200

// ----- Individual trend and pullback quality -----
perf3m    = close / close[63] - 1.0
perf21    = close / close[21] - 1.0
ema50Dist = close / ema50 - 1.0

trendOk = close > ema200 and ema50 > ema200 and perf3m > 0
pullbackOk = perf21 <= -0.03 and ema50Dist >= -0.01 and volRatio >= minVolumeRatio

// ----- Price action: 4 of 7 -----
barRange  = math.max(high - low, syminfo.mintick)
lowerWick = math.min(open, close) - low

pa1 = close > open
pa2 = lowerWick / barRange >= 0.40
pa3 = (close - low) / barRange >= 0.70
pa4 = low > low[1] and low[1] > low[2]
pa5 = close > high[1]
pa6 = high[1] <= high[2] and low[1] >= low[2] and close > high[1]
pa7 = close > open[1] and open < close[1] and close > open

paCount = (pa1 ? 1 : 0) + (pa2 ? 1 : 0) + (pa3 ? 1 : 0) +
     (pa4 ? 1 : 0) + (pa5 ? 1 : 0) + (pa6 ? 1 : 0) + (pa7 ? 1 : 0)

// ----- Oscillators: 1 of 3 -----
osc1 = rsi14 > rsi14[1]
osc2 = ta.crossover(stochK, stochD)
osc3 = macdHist > macdHist[1]
oscCount = (osc1 ? 1 : 0) + (osc2 ? 1 : 0) + (osc3 ? 1 : 0)

reversalOk = paCount >= minPaCount and oscCount >= minOscCount

// ----- Technical stop and resistance target -----
plannedStop   = ta.lowest(low, stopLookback) - atrBuffer * atr14
plannedTarget = ta.highest(high[1], targetLookback) * 0.995
plannedRisk   = close - plannedStop
plannedRR     = plannedRisk > 0 ? (plannedTarget - close) / plannedRisk : na

baseSignal = barstate.isconfirmed and marketOk and trendOk and pullbackOk and reversalOk
provisionalRR = plannedRR >= minRR and plannedRR <= maxRR
entrySignal = baseSignal and provisionalRR and strategy.position_size == 0

// Orders submitted at the signal close fill at the next available open.
var float activeStop = na
var float activeTarget = na
var int entryBar = na

if entrySignal
    activeStop := plannedStop
    activeTarget := plannedTarget
    strategy.entry("Long", strategy.long)

if strategy.position_size > 0 and strategy.position_size[1] <= 0
    entryBar := bar_index

// Actual RR is displayed after the next bar opens.
actualRisk = strategy.position_size > 0 ? strategy.position_avg_price - activeStop : na
openRR = strategy.position_size > 0 and actualRisk > 0 ?
     (activeTarget - strategy.position_avg_price) / actualRisk : na

if strategy.position_size > 0
    strategy.exit("Structure Exit", "Long", stop=activeStop, limit=activeTarget)
    if bar_index - entryBar >= maxHolding
        strategy.close("Long", comment="Time exit")

// ----- Visuals -----
plot(ema20, "EMA20", color=color.new(color.teal, 0))
plot(ema50, "EMA50", color=color.new(color.orange, 0))
plot(ema200, "EMA200", color=color.new(color.red, 0), linewidth=2)
plot(strategy.position_size > 0 ? activeStop : na, "Stop", color=color.red,
     style=plot.style_linebr, linewidth=2)
plot(strategy.position_size > 0 ? activeTarget : na, "60D Resistance Target",
     color=color.lime, style=plot.style_linebr, linewidth=2)

plotshape(baseSignal, title="Setup", style=shape.circle, location=location.belowbar,
     color=provisionalRR ? color.lime : color.orange, size=size.tiny, text="RR")

bgcolor(not marketOk ? color.new(color.red, 92) : na, title="Market gate off")

// Data-window values
plot(plannedRR, "Planned RR", display=display.data_window)
plot(openRR, "Actual Open RR", display=display.data_window)
plot(paCount, "PA Count", display=display.data_window)
plot(oscCount, "Osc Count", display=display.data_window)
plot(perf21 * 100, "21D Return %", display=display.data_window)
plot(ema50Dist * 100, "EMA50 Distance %", display=display.data_window)
plot(volRatio, "Volume Ratio", display=display.data_window)

alertcondition(baseSignal, title="RR Pullback Setup",
     message="Pullback reversal setup. Recalculate RR at next open; enter only at RR 1.5-2.0.")
